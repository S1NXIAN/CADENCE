    19→/**
    20→ * Coach engine v2 — "observant, not generic".
    21→ *
    22→ * The old coach only looked at aggregate scores (accuracy, wpm, consistency).
    23→ * This version also reads the two richest sources the app already records:
    24→ *
    25→ *   1. the keystroke event log — which exact letters slipped, which targeted
    26→ *      keys failed, where the fingers froze, which keys ran slow THIS test;
    27→ *   2. the per-second samples — fade (stamina), cold opens (warmup), burst
    28→ *      spread (rhythm), and error clusters ("second 14 is where it went").
    29→ *
    30→ * Detectors produce prioritized candidates; the selector keeps the most
    31→ * urgent notes with kind caps so the panel never shows four variations of
    32→ * the same observation. Phrase variants rotate deterministically per test
    33→ * (hash of result id) so consecutive similar tests don't read identically.
    34→ */
    35→
    36→// ---------------------------------------------------------------------------
    37→// Per-test keystroke forensics
    38→// ---------------------------------------------------------------------------
    39→
    40→const HESITATION_MS = 900;
    41→const LAT_MAX = 2000; // pauses longer than this are thinking, not typing
    42→const TRACK_RE = /[a-z0-9';.,!?-]/i;
    43→
    44→interface EventAnalysis {
    45→  keystrokes: number;
    46→  errors: number;
    47→  missed: number; // expected chars that never landed (synthetic "" events)
    48→  extra: number; // chars typed past the end of a word
    49→  confusions: Map<string, number>; // "e>r" -> count, this test
    50→  focusSlips: Map<string, number>; // focus key -> error count, this test
    51→  latencies: Map<string, number[]>; // pressed key -> ms samples, this test
    52→  hesitations: Array<{ ch: string; ms: number }>; // pauses >= HESITATION_MS
    53→}
    54→
    55→function analyzeEvents(events: CharEvent[], focusKeys: string[]): EventAnalysis {
    56→  const a: EventAnalysis = {
    57→    keystrokes: events.length,
    58→    errors: 0,
    59→    missed: 0,
    60→    extra: 0,
    61→    confusions: new Map(),
    62→    focusSlips: new Map(),
    63→    latencies: new Map(),
    64→    hesitations: [],
    65→  };
    66→  const focus = new Set(focusKeys);
    67→  let prevT: number | null = null;
    68→  let prevCorrect = false;
    69→
    70→  for (const ev of events) {
    71→    const typedTracked = ev.typed.length === 1 && TRACK_RE.test(ev.typed);
    72→
    73→    // inter-key latency on clean presses (mirrors profiles.ingestEvents rules)
    74→    if (ev.correct && typedTracked && prevT !== null && prevCorrect) {
    75→      const d = ev.t - prevT;
    76→      if (d > 20 && d < LAT_MAX) {
    77→        const k = ev.typed.toLowerCase();
    78→        const arr = a.latencies.get(k);
    79→        if (arr) arr.push(d);
    80→        else a.latencies.set(k, [d]);
    81→      }
    82→      if (d >= HESITATION_MS) a.hesitations.push({ ch: ev.typed, ms: d });
    83→    }
    84→
    85→    if (!ev.correct) {
    86→      a.errors += 1;
    87→      if (ev.typed === "" && ev.expected) {
    88→        // dropped letter — blame the expected key, including focus keys
    89→        a.missed += 1;
    90→        const k = ev.expected.toLowerCase();
    91→        if (focus.has(k)) a.focusSlips.set(k, (a.focusSlips.get(k) ?? 0) + 1);
    92→      }
    93→    } else if (ev.expected === null) {
    94→      a.extra += 1;
    95→    }
    96→
    97→    // confusion pair (same eligibility rules as profiles.ingestEvents)
    98→    if (
    99→      ev.expected &&
   100→      ev.typed.length === 1 &&
   101→      ev.expected.toLowerCase() !== ev.typed.toLowerCase() &&
   102→      TRACK_RE.test(ev.expected) &&
   103→      TRACK_RE.test(ev.typed)
   104→    ) {
   105→      const key = `${ev.expected.toLowerCase()}>${ev.typed.toLowerCase()}`;
   106→      a.confusions.set(key, (a.confusions.get(key) ?? 0) + 1);
   107→      const k = ev.expected.toLowerCase();
   108→      if (focus.has(k)) a.focusSlips.set(k, (a.focusSlips.get(k) ?? 0) + 1);
   109→    }
   110→
   111→    prevT = ev.t;
   112→    prevCorrect = ev.correct;
   113→  }
   114→  return a;
   115→}

   121→interface CurveAnalysis {
   122→  firstMean: number; // mean instantaneous wpm, first 35% of active seconds
   123→  lastMean: number; // same, last 35%
   124→  first3: number; // mean of the first 3 active seconds (cold-open signal)
   125→  overallMean: number;
   126→  fade: number; // (first - last) / first, >= 0 — stamina signal
   127→  warmup: number; // (overall - first3) / overall — cold-open signal
   128→  peak: number;
   129→  peakSecond: number;
   130→  low: number; // slowest active second
   131→  errTotal: number;
   132→  worstSec: number;
   133→  worstSecErrs: number;
   134→  worstWindowStart: number; // worst 3-second error window
   135→  worstWindowErrs: number;
   136→}
   137→
   138→function analyzeCurve(result: TestResult): CurveAnalysis | null {
   139→  const s = result.samples;
   140→  if (s.length < 4) return null;
   141→
   142→  // cumulative raw → per-second instantaneous wpm:
   143→  // chars(k) = raw(k)*k/12 (cumulative), inst(k) = raw(k)*k - raw(k-1)*(k-1)
   144→  const instAll: number[] = [];
   145→  for (let i = 0; i < s.length; i++) {
   146→    const cum = s[i].raw * s[i].second;
   147→    const prev = i > 0 ? s[i - 1].raw * s[i - 1].second : 0;
   148→    instAll.push(Math.max(0, cum - prev));
   149→  }
   150→  const act: number[] = [];
   151→  const actSec: number[] = [];
   152→  for (let i = 0; i < instAll.length; i++) {
   153→    if (instAll[i] > 0) {
   154→      act.push(instAll[i]);
   155→      actSec.push(s[i].second);
   156→    }
   157→  }
   158→  if (act.length < 4) return null;
   159→
   160→  const mean = (xs: number[]) => xs.reduce((x, y) => x + y, 0) / xs.length;
   161→  const overallMean = mean(act);
   162→  const edge = Math.max(2, Math.floor(act.length * 0.35));
   163→  const firstMean = mean(act.slice(0, edge));
   164→  const lastMean = mean(act.slice(act.length - edge));
   165→  const first3 = act.length >= 3 ? mean(act.slice(0, 3)) : firstMean;
   166→  const fade = firstMean > 20 ? Math.max(0, (firstMean - lastMean) / firstMean) : 0;
   167→  const warmup = overallMean > 20 ? Math.max(0, (overallMean - first3) / overallMean) : 0;
   168→
   169→  let peak = 0;
   170→  let peakSecond = 0;
   171→  let low = Infinity;
   172→  for (let i = 0; i < act.length; i++) {
   173→    if (act[i] > peak) {
   174→      peak = act[i];
   175→      peakSecond = actSec[i];
   176→    }
   177→    if (act[i] < low) low = act[i];
   178→  }
   179→
   180→  let errTotal = 0;
   181→  let worstSec = 0;
   182→  let worstSecErrs = 0;
   183→  for (const x of s) {
   184→    errTotal += x.errors;
   185→    if (x.errors > worstSecErrs) {
   186→      worstSecErrs = x.errors;
   187→      worstSec = x.second;
   188→    }
   189→  }
   190→  let worstWindowStart = 0;
   191→  let worstWindowErrs = 0;
   192→  for (let i = 0; i < s.length; i++) {
   193→    let sum = 0;
   194→    for (let j = i; j < Math.min(s.length, i + 3); j++) sum += s[j].errors;
   195→    if (sum > worstWindowErrs) {
   196→      worstWindowErrs = sum;
   197→      worstWindowStart = s[i].second;
   198→    }
   199→  }
   200→
   201→  return {
   202→    firstMean,
   203→    lastMean,
   204→    first3,
   205→    overallMean,
   206→    fade,
   207→    warmup,
   208→    peak,
   209→    peakSecond,
   210→    low,
   211→    errTotal,
   212→    worstSec,
   213→    worstSecErrs,
   214→    worstWindowStart,
   215→    worstWindowErrs,
   216→  };
   217→}

   240→function hashStr(s: string): number {
   241→  let h = 0;
   242→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
   243→  return Math.abs(h);
   244→}

   246→function recentSameLabel(stats: StatsData, label: string, excludeId: string): TestResult[] {
   247→  return stats.history.filter((h) => h.modeLabel === label && h.id !== excludeId);

   250→/** record: new PB with delta, first baseline, or a near-miss. */
   251→function recordNote(c: DetectorCtx): Candidate | null {
   252→  const { result, v } = c;
   253→  if (result.wpm <= 0) return null;
   254→  const prev = c.prevBest !== undefined ? c.prevBest : c.stats.personalBests[result.modeLabel];
   255→  if (result.isPersonalBest) {
   256→    if (prev && result.wpm > prev.wpm) {
   257→      return {
   258→        priority: 85,
   259→        note: {
   260→          kind: "record",
   261→          title: v(["new personal best", "record falls"]),
   262→          message: v([
   263→            `${result.wpm} WPM for ${result.modeLabel}, up ${result.wpm - prev.wpm} from your previous best. Speed follows accuracy — you just proved it again.`,
   264→            `+${result.wpm - prev.wpm} over your old ${result.modeLabel} best. Whatever pace you held in the clean middle stretch — that's the new default to train at.`,
   265→          ]),
   266→          metric: `+${result.wpm - prev.wpm} wpm`,
   267→        },
   268→      };
   269→    }
   270→    return {
   271→      priority: 85,
   272→      note: {
   273→        kind: "record",
   274→        title: "baseline set",
   275→        message: `First score on record for ${result.modeLabel}: ${result.wpm} WPM at ${result.accuracy}%. Every test from here is measured against it.`,
   276→        metric: `${result.wpm} wpm`,
   277→      },
   278→    };
   279→  }
   280→  if (prev && prev.wpm - result.wpm <= 2 && prev.wpm > result.wpm) {
   281→    return {
   282→      priority: 55,
   283→      note: {
   284→        kind: "record",
   285→        title: "knocking on the record",
   286→        message: `${prev.wpm - result.wpm === 0 ? "Tied pace with" : `Just ${prev.wpm - result.wpm} under`} your ${result.modeLabel} best (${prev.wpm}). One cleaner run today takes it.`,
   287→        metric: `best ${prev.wpm}`,
   288→      },
   289→    };
   290→  }
   291→  return null;
   292→}

   316→  if (result.accuracy < 92) {
   317→    if (result.chars.incorrect + result.chars.extra <= 3) return null;
   318→    return {
   319→      priority: 92,
   320→      note: {
   321→        kind: "accuracy",
   322→        title: v(["accuracy below threshold", "sloppy reps detected"]),

   325→          `At ${result.accuracy}%, every rep is partly practice for the WRONG motion. Slow down until you're back above 95; speed re-earns itself fast.`,
   326→        ]),
   327→        metric: `${result.accuracy}%`,
   328→      },
   329→    };
   330→  }
   331→  if (result.accuracy < 96 && ev.errors >= 4) {

   333→      priority: 64,

   335→        kind: "accuracy",
   336→        title: "nearly clean",

   338→        metric: `${ev.errors} slips`,
   339→      },
   340→    };
   341→  }
   342→  if (result.accuracy >= 98 && result.wpm > 0) {

   346→        kind: "accuracy",
   347→        title: v(["elite precision", "surgical run"]),

   352→        metric: `${result.accuracy}%`,
   353→      },
   354→    };
   355→  }
   356→  return null;

   359→/** trend: wpm vs last runs of the same label. */
   360→function wpmTrendNote(c: DetectorCtx): Candidate | null {
   361→  const { result, v } = c;
   362→  const same = recentSameLabel(c.stats, result.modeLabel, result.id);
   363→  if (same.length < 3) return null;
   364→  const last5 = same.slice(0, 5);
   365→  const avg = last5.reduce((a, b) => a + b.wpm, 0) / last5.length;
   366→  const delta = Math.round(result.wpm - avg);
   367→  if (delta >= 3) {
   368→    return {
   369→      priority: 60,
   370→      note: {
   371→        kind: "trend",
   372→        title: v(["above your curve", "trending up"]),
   373→        message: v([
   374→          `+${delta} WPM over your last ${last5.length} ${result.modeLabel} runs (avg ${Math.round(avg)}). The training is compounding — same time tomorrow keeps the curve bent.`,
   375→          `${delta} wpm better than your recent ${Math.round(avg)} average. Progress like this is made of boring, repeated, accurate reps — keep stacking them.`,
   376→        ]),
   377→        metric: `${delta >= 0 ? "+" : ""}${delta} wpm`,
   378→      },
   379→    };
   380→  }
   381→  if (delta <= -5) {
   382→    return {
   383→      priority: 62,
   384→      note: {
   385→        kind: "trend",
   386→        title: "off-pace day",
   387→        message: `${delta} WPM vs your recent ${Math.round(avg)} average. Off days are data: check whether accuracy dipped or one key cluster slowed the whole run.`,
   388→        metric: `${delta} wpm`,
   389→      },
   390→    };
   391→  }
   392→  return null;
   393→}

   490→/** focus: did the keys this test targeted actually fail or hold? */
   491→function focusNote(c: DetectorCtx): Candidate | null {
   492→  const { result, ev } = c;
   493→  if (result.focusKeys.length === 0) return null;
   494→  const slips = [...ev.focusSlips.entries()].sort((a, b) => b[1] - a[1]);
   495→  const worst = slips[0];
   496→  if (worst && worst[1] >= 2) {
   497→    return {
   498→      priority: 76,
   499→      note: {
   500→        kind: "focus",
   501→        title: "targeted key still slipping",
   502→        message: `'${worst[0]}' — a key this test was built around — slipped ${worst[1]}×. That's the point: the generator feeds you your weakness until it's automatic. Next round features it again.`,
   503→        metric: `'${worst[0]}' ×${worst[1]}`,
   504→      },
   505→    };
   506→  }
   507→  if (slips.length === 0 && ev.keystrokes >= 40) {
   508→    return {
   509→      priority: 44,
   510→      note: {
   511→        kind: "recovery",
   512→        title: "targets held clean",
   513→        message: `Every key this test drilled (${result.focusKeys.map((k) => `'${k}'`).join(", ")}) stayed error-free. Clean exposure is how weak keys graduate — the engine will rotate in the next bottleneck.`,
   514→      },
   515→    };
   516→  }
   517→  return null;
   518→}

   520→/** error: dropped letters vs overshoot diagnosis. */
   521→function charLossNote(c: DetectorCtx): Candidate | null {
   522→  const { result, v } = c;
   523→  const { missed, extra } = result.chars;
   524→  if (missed >= 4 && missed >= extra * 2) {
   525→    return {
   526→      priority: 66,
   527→      note: {
   528→        kind: "error",
   529→        title: v(["dropping letters", "truncated words"]),
   530→        message: v([
   531→          `${missed} expected characters never landed (vs ${extra} extras). That signature means releasing a word before its last letter — finish every word to the final character, THEN hit space.`,
   532→          `${missed} letters silently skipped. Fingers are finishing early — a beat of patience on each word's tail letters erases this whole class of error.`,
   533→        ]),
   534→        metric: `${missed} missed`,
   535→      },
   536→    };
   537→  }
   538→  if (extra >= 4 && extra > missed) {
   539→    return {
   540→      priority: 62,
   541→      note: {
   542→        kind: "error",
   543→        title: "overshooting words",
   544→        message: `${extra} extra characters landed past word ends. Anticipating fingers type the letter they EXPECT next — let the target word finish before the space bar does.`,
   545→        metric: `${extra} extra`,
   546→      },
   547→    };
   548→  }
   549→  return null;
   550→}

   552→/** tip: the sharpest finger habit, lifetime + this test combined. */
   553→function confusionNote(c: DetectorCtx): Candidate | null {
   554→  const { ev, learning, v } = c;
   555→  const combined = new Map<string, { expected: string; typed: string; total: number; thisTest: number }>();
   556→  for (const c2 of learning.confusions) {
   557→    const key = `${c2.expected}>${c2.typed}`;
   558→    combined.set(key, { expected: c2.expected, typed: c2.typed, total: c2.count, thisTest: 0 });
   559→  }
   560→  for (const [key, n] of ev.confusions) {
   561→    const [e, t] = key.split(">");
   562→    const cur = combined.get(key);
   563→    if (cur) {
   564→      cur.total += n;
   565→      cur.thisTest += n;
   566→    } else {
   567→      combined.set(key, { expected: e, typed: t, total: n, thisTest: n });
   568→    }
   569→  }
   570→  let best: { expected: string; typed: string; total: number; thisTest: number } | null = null;
   571→  for (const x of combined.values()) {
   572→    if (x.total < 3 && x.thisTest < 2) continue;
   573→    if (!best || x.total > best.total || (x.total === best.total && x.thisTest > best.thisTest)) best = x;
   574→  }
   575→  if (!best) return null;
   576→  return {
   577→    priority: 70,
   578→    note: {
   579→      kind: "tip",
   580→      title: v([`finger habit: '${best.expected}' → '${best.typed}'`, `'${best.expected}' keeps turning into '${best.typed}'`]),
   581→      message: v([
   582→        `You've swapped '${best.expected}' for '${best.typed}' ${best.total}× recently${best.thisTest > 0 ? `, ${best.thisTest} in this very test` : ""}. That's a fixed finger-path habit — slow, deliberate reps on exactly that pair rewire it faster than volume does.`,
   583→        `The scorecard says '${best.expected}' becomes '${best.typed}' under pressure (${best.total}×). Isolate it: type ten slow words containing '${best.expected}' before the next run.`,
   584→      ]),
   585→      metric: `×${best.total}${best.thisTest ? ` · ${best.thisTest} now` : ""}`,
   586→    },
   587→  };
   588→}

   590→/** tip: trigram error context — the transition, not the key. */
   591→function transitionNote(c: DetectorCtx): Candidate | null {
   592→  const ctx = topErrorContexts(c.learning, 1)[0];
   593→  if (!ctx) return null;
   594→  const before = ctx.trigram.slice(0, 2);
   595→  const target = ctx.trigram.slice(2);
   596→  return {
   597→    priority: 58,
   598→    note: {
   599→      kind: "tip",
   600→      title: `transition '${before}' → '${target}'`,
   601→      message: `Your errors cluster right after '${before}' (${ctx.count}× recently). The combination is what trips you, not the lone key — adaptive mode drills that exact sequence.`,
   602→      metric: `×${ctx.count}`,
   603→    },
   604→  };
   605→}

   607→/** tip: slowest keys THIS test vs personal baseline. */
   608→function slowKeysNote(c: DetectorCtx): Candidate | null {
   609→  const { ev, learning } = c;
   610→  const median = personalMedianLatency(learning) || 150;
   611→  const ranked: Array<{ key: string; ms: number; ratio: number }> = [];
   612→  for (const [key, arr] of ev.latencies) {
   613→    if (arr.length < 3) continue;
   614→    const sorted = [...arr].sort((a, b) => a - b);
   615→    const med = sorted[Math.floor(sorted.length / 2)];
   616→    const ratio = med / median;
   617→    if (ratio >= 1.35) ranked.push({ key, ms: med, ratio });
   618→  }
   619→  if (ranked.length === 0) return null;
   620→  ranked.sort((a, b) => b.ratio - a.ratio);
   621→  const top = ranked.slice(0, 2);
   622→  return {
   623→    priority: 48,
   624→    note: {
   625→      kind: "tip",
   626→      title: "slowest keys this run",
   627→      message: `${top.map((k) => `'${k.key}' (${k.ms}ms)`).join(" and ")} ran well above your ~${Math.round(median)}ms baseline. They're pace-killers, not error-makers — deliberate reps on words containing them pays double.`,
   628→      metric: top.map((k) => `'${k.key}' ${k.ms}ms`).join(" "),
   629→    },
   630→  };
   631→}

   667→/** recovery: keys that used to cost errors are now clean. */
   668→function recoveredNote(c: DetectorCtx): Candidate | null {
   669→  const { learning } = c;
   670→  const now = Date.now();
   671→  const healed: Array<{ key: string; r: number }> = [];
   672→  for (const [key, p] of Object.entries(learning.keyProfiles)) {
   673→    if (p.attempts < 20 || p.errors < 2) continue;
   674→    if (p.errRate > 0.02) continue;
   675→    if (!p.mem || p.mem.reps === 0) continue;
   676→    const r = memRetrievability(p.mem, now);
   677→    if (r >= 0.8) healed.push({ key, r });
   678→  }
   679→  if (healed.length < 2) return null;
   680→  healed.sort((a, b) => b.r - a.r);
   681→  const top = healed.slice(0, 2);
   682→  return {
   683→    priority: 45,
   684→    note: {
   685→      kind: "recovery",
   686→      title: "off the watch list",
   687→      message: `'${top.map((k) => k.key).join("', '")}' used to cost you errors — recent runs are clean and the memory model holds them at ${Math.round(top[0].r * 100)}% reliability. The generator will fade them out of drills and hunt the next bottleneck.`,
   688→      metric: `${Math.round(top[0].r * 100)}% solid`,
   689→    },
   690→  };
   691→}

   750→
   751→/**
   752→ * Generate human coaching insights after a completed test.
   753→ * This is the "wants you to improve" personality of the app.
   754→ *
   755→ * `events` is the raw keystroke log of the finished test — the coach reads it
   756→ * directly (which letters slipped, where pauses fell) instead of judging from
   757→ * aggregates alone. `stats` should be the POST-record stats (so streak/history
   758→ * notes reflect the test that just finished); `prevBest` is the personal best
   759→ * BEFORE this test was recorded — pass it so the "new PB +delta" math works
   760→ * (after recording, the PB already IS this result, and the delta would read 0).
   761→ */
   762→export function generateInsights(
   763→  result: TestResult,
   764→  events: CharEvent[],
   765→  stats: StatsData,
   766→  learning: LearningData,
   767→  settings?: Settings,
   768→  prevBest?: PersonalBest | null
   769→): CoachInsight[] {
   770→  const ev = analyzeEvents(events, result.focusKeys);
   771→  const curve = analyzeCurve(result);
   772→  const h = hashStr(result.id);
   773→  const ctx: DetectorCtx = {
   774→    result,
   775→    ev,
   776→    curve,
   777→    stats,
   778→    learning,
   779→    settings,
   780→    prevBest,
   781→    v: <T,>(arr: T[]) => arr[h % arr.length],
   782→  };
   783→
   784→  const candidates = [
   785→    recordNote(ctx),
   786→    errorTaxNote(ctx),
   787→    accuracyNote(ctx),
   788→    wpmTrendNote(ctx),
   789→    accuracyTrendNote(ctx),
   790→    fadeNote(ctx),
   791→    warmupNote(ctx),
   792→    errorSpikeNote(ctx),
   793→    spreadNote(ctx),
   794→    focusNote(ctx),
   795→    charLossNote(ctx),
   796→    confusionNote(ctx),
   797→    transitionNote(ctx),
   798→    slowKeysNote(ctx),
   799→    hesitationNote(ctx),
   800→    speedPushNote(ctx),
   801→    recoveredNote(ctx),
   802→    dueNote(ctx),
   803→    streakNote(ctx),
   804→    adaptiveNudgeNote(ctx),
   805→  ].filter((x): x is Candidate => x !== null && x.note.message.length > 0);
   806→
   807→  candidates.sort((a, b) => b.priority - a.priority);
   808→
   809→  // kind caps: one note per kind (tips may double up), max 4 total, no dupes
   810→  const picked: Candidate[] = [];
   811→  const seen = new Set<string>();
   812→  const kindCount = new Map<CoachInsight["kind"], number>();
   813→  for (const cand of candidates) {
   814→    const sig = cand.note.message;
   815→    if (seen.has(sig)) continue;
   816→    const n = kindCount.get(cand.note.kind) ?? 0;
   817→    const cap = cand.note.kind === "tip" ? 2 : 1;
   818→    if (n >= cap) continue;
   819→    kindCount.set(cand.note.kind, n + 1);
   820→    seen.add(sig);
   821→    picked.push(cand);
   822→    if (picked.length >= 4) break;
   823→  }
   824→  return picked.map((p) => p.note);
   825→}

   831→/** Idle-state line: what the engine will target next (adaptive mode only). */
   832→export function nextTestPreview(learning: LearningData, mode: string): string {
   833→  if (mode !== "adaptive") return "";
   834→  if (learning.totalTests < 1) {
   835→    return "Your first test calibrates the coach. From test 2 onward, every word is chosen for you.";
   836→  }
   837→  const weak = computeWeakKeys(learning).filter((w) => w.weakness > 0.08).slice(0, 2);
   838→  const bgs = weakBigrams(learning, 3).filter((b) => b.weakness > 0.08).slice(0, 2);
   839→
   840→  const horizon = Date.now() + 48 * 3600 * 1000;
   841→  let due = 0;
   842→  for (const p of Object.values(learning.keyProfiles)) {
   843→    if (p.mem && p.mem.reps > 0 && p.mem.due <= horizon) due += 1;
   844→  }
   845→  for (const p of Object.values(learning.bigramProfiles)) {
   846→    if (p.mem && p.mem.reps > 0 && p.mem.due <= horizon) due += 1;
   847→  }
   848→
   849→  const parts: string[] = [];
   850→  if (weak.length) parts.push(`keys ${weak.map((w) => `'${w.key}'`).join(", ")}`);
   851→  if (bgs.length) parts.push(`transitions ${bgs.map((b) => `'${b.bigram}'`).join(", ")}`);
   852→
   853→  if (!parts.length) {
   854→    if (due >= 4) {
   855→      return `${due} items are coming due for a refresher — the next test will start resurfacing them before they fade.`;
   856→    }
   857→    return "No strong weaknesses right now — the engine keeps mapping your fingers.";
   858→  }
   859→  const suffix = due >= 4 ? `, plus ${due} items due for a refresher` : "";
   860→  return `Next test attacks your ${parts.join(" and ")}${suffix}.`;
   861→}
