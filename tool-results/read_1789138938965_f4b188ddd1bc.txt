    19→    19→/**
    20→    20→ * Coach engine v2 — "observant, not generic".
    21→    21→ *
    22→    22→ * The old coach only looked at aggregate scores (accuracy, wpm, consistency).
    23→    23→ * This version also reads the two richest sources the app already records:
    24→    24→ *
    25→    25→ *   1. the keystroke event log — which exact letters slipped, which targeted
    26→    26→ *      keys failed, where the fingers froze, which keys ran slow THIS test;
    27→    27→ *   2. the per-second samples — fade (stamina), cold opens (warmup), burst
    28→    28→ *      spread (rhythm), and error clusters ("second 14 is where it went").
    29→    29→ *
    30→    30→ * Detectors produce prioritized candidates; the selector keeps the most
    31→    31→ * urgent notes with kind caps so the panel never shows four variations of
    32→    32→ * the same observation. Phrase variants rotate deterministically per test
    33→    33→ * (hash of result id) so consecutive similar tests don't read identically.
    34→    34→ */

    36→    36→// ---------------------------------------------------------------------------
    37→    37→// Per-test keystroke forensics
    38→    38→// ---------------------------------------------------------------------------
    39→    39→
    40→    40→const HESITATION_MS = 900;
    41→    41→const LAT_MAX = 2000; // pauses longer than this are thinking, not typing
    42→    42→const TRACK_RE = /[a-z0-9';.,!?-]/i;
    43→    43→
    44→    44→interface EventAnalysis {
    45→    45→  keystrokes: number;
    46→    46→  errors: number;
    47→    47→  missed: number; // expected chars that never landed (synthetic "" events)
    48→    48→  extra: number; // chars typed past the end of a word
    49→    49→  confusions: Map<string, number>; // "e>r" -> count, this test
    50→    50→  focusSlips: Map<string, number>; // focus key -> error count, this test
    51→    51→  latencies: Map<string, number[]>; // pressed key -> ms samples, this test
    52→    52→  hesitations: Array<{ ch: string; ms: number }>; // pauses >= HESITATION_MS
    53→    53→}
    54→    54→
    55→    55→function analyzeEvents(events: CharEvent[], focusKeys: string[]): EventAnalysis {
    56→    56→  const a: EventAnalysis = {
    57→    57→    keystrokes: events.length,
    58→    58→    errors: 0,
    59→    59→    missed: 0,
    60→    60→    extra: 0,
    61→    61→    confusions: new Map(),
    62→    62→    focusSlips: new Map(),
    63→    63→    latencies: new Map(),
    64→    64→    hesitations: [],
    65→    65→  };
    66→    66→  const focus = new Set(focusKeys);
    67→    67→  let prevT: number | null = null;
    68→    68→  let prevCorrect = false;
    69→    69→
    70→    70→  for (const ev of events) {
    71→    71→    const typedTracked = ev.typed.length === 1 && TRACK_RE.test(ev.typed);
    72→    72→
    73→    73→    // inter-key latency on clean presses (mirrors profiles.ingestEvents rules)
    74→    74→    if (ev.correct && typedTracked && prevT !== null && prevCorrect) {
    75→    75→      const d = ev.t - prevT;
    76→    76→      if (d > 20 && d < LAT_MAX) {
    77→    77→        const k = ev.typed.toLowerCase();
    78→    78→        const arr = a.latencies.get(k);
    79→    79→        if (arr) arr.push(d);
    80→    80→        else a.latencies.set(k, [d]);
    81→    81→      }
    82→    82→      if (d >= HESITATION_MS) a.hesitations.push({ ch: ev.typed, ms: d });
    83→    83→    }
    84→    84→
    85→    85→    if (!ev.correct) {
    86→    86→      a.errors += 1;
    87→    87→      if (ev.typed === "" && ev.expected) {
    88→    88→        // dropped letter — blame the expected key, including focus keys
    89→    89→        a.missed += 1;
    90→    90→        const k = ev.expected.toLowerCase();
    91→    91→        if (focus.has(k)) a.focusSlips.set(k, (a.focusSlips.get(k) ?? 0) + 1);
    92→    92→      }
    93→    93→    } else if (ev.expected === null) {
    94→    94→      a.extra += 1;
    95→    95→    }
    96→    96→
    97→    97→    // confusion pair (same eligibility rules as profiles.ingestEvents)
    98→    98→    if (
    99→    99→      ev.expected &&
   100→   100→      ev.typed.length === 1 &&
   101→   101→      ev.expected.toLowerCase() !== ev.typed.toLowerCase() &&
   102→   102→      TRACK_RE.test(ev.expected) &&
   103→   103→      TRACK_RE.test(ev.typed)
   104→   104→    ) {
   105→   105→      const key = `${ev.expected.toLowerCase()}>${ev.typed.toLowerCase()}`;
   106→   106→      a.confusions.set(key, (a.confusions.get(key) ?? 0) + 1);
   107→   107→      const k = ev.expected.toLowerCase();
   108→   108→      if (focus.has(k)) a.focusSlips.set(k, (a.focusSlips.get(k) ?? 0) + 1);
   109→   109→    }
   110→   110→
   111→   111→    prevT = ev.t;
   112→   112→    prevCorrect = ev.correct;
   113→   113→  }
   114→   114→  return a;
   115→   115→}

   121→   121→interface CurveAnalysis {
   122→   122→  firstMean: number; // mean instantaneous wpm, first 35% of active seconds
   123→   123→  lastMean: number; // same, last 35%
   124→   124→  first3: number; // mean of the first 3 active seconds (cold-open signal)
   125→   125→  overallMean: number;
   126→   126→  fade: number; // (first - last) / first, >= 0 — stamina signal
   127→   127→  warmup: number; // (overall - first3) / overall — cold-open signal
   128→   128→  peak: number;
   129→   129→  peakSecond: number;
   130→   130→  low: number; // slowest active second
   131→   131→  errTotal: number;
   132→   132→  worstSec: number;
   133→   133→  worstSecErrs: number;
   134→   134→  worstWindowStart: number; // worst 3-second error window
   135→   135→  worstWindowErrs: number;
   136→   136→}
   137→   137→
   138→   138→function analyzeCurve(result: TestResult): CurveAnalysis | null {
   139→   139→  const s = result.samples;
   140→   140→  if (s.length < 4) return null;
   141→   141→
   142→   142→  // cumulative raw → per-second instantaneous wpm:
   143→   143→  // chars(k) = raw(k)*k/12 (cumulative), inst(k) = raw(k)*k - raw(k-1)*(k-1)
   144→   144→  const instAll: number[] = [];
   145→   145→  for (let i = 0; i < s.length; i++) {
   146→   146→    const cum = s[i].raw * s[i].second;
   147→   147→    const prev = i > 0 ? s[i - 1].raw * s[i - 1].second : 0;
   148→   148→    instAll.push(Math.max(0, cum - prev));
   149→   149→  }
   150→   150→  const act: number[] = [];
   151→   151→  const actSec: number[] = [];
   152→   152→  for (let i = 0; i < instAll.length; i++) {
   153→   153→    if (instAll[i] > 0) {
   154→   154→      act.push(instAll[i]);
   155→   155→      actSec.push(s[i].second);
   156→   156→    }
   157→   157→  }
   158→   158→  if (act.length < 4) return null;
   159→   159→
   160→   160→  const mean = (xs: number[]) => xs.reduce((x, y) => x + y, 0) / xs.length;
   161→   161→  const overallMean = mean(act);
   162→   162→  const edge = Math.max(2, Math.floor(act.length * 0.35));
   163→   163→  const firstMean = mean(act.slice(0, edge));
   164→   164→  const lastMean = mean(act.slice(act.length - edge));
   165→   165→  const first3 = act.length >= 3 ? mean(act.slice(0, 3)) : firstMean;
   166→   166→  const fade = firstMean > 20 ? Math.max(0, (firstMean - lastMean) / firstMean) : 0;
   167→   167→  const warmup = overallMean > 20 ? Math.max(0, (overallMean - first3) / overallMean) : 0;
   168→   168→
   169→   169→  let peak = 0;
   170→   170→  let peakSecond = 0;
   171→   171→  let low = Infinity;
   172→   172→  for (let i = 0; i < act.length; i++) {
   173→   173→    if (act[i] > peak) {
   174→   174→      peak = act[i];
   175→   175→      peakSecond = actSec[i];
   176→   176→    }
   177→   177→    if (act[i] < low) low = act[i];
   178→   178→  }
   179→   179→
   180→   180→  let errTotal = 0;
   181→   181→  let worstSec = 0;
   182→   182→  let worstSecErrs = 0;
   183→   183→  for (const x of s) {
   184→   184→    errTotal += x.errors;
   185→   185→    if (x.errors > worstSecErrs) {
   186→   186→      worstSecErrs = x.errors;
   187→   187→      worstSec = x.second;
   188→   188→    }
   189→   189→  }
   190→   190→  let worstWindowStart = 0;
   191→   191→  let worstWindowErrs = 0;
   192→   192→  for (let i = 0; i < s.length; i++) {
   193→   193→    let sum = 0;
   194→   194→    for (let j = i; j < Math.min(s.length, i + 3); j++) sum += s[j].errors;
   195→   195→    if (sum > worstWindowErrs) {
   196→   196→      worstWindowErrs = sum;
   197→   197→      worstWindowStart = s[i].second;
   198→   198→    }
   199→   199→  }
   200→   200→
   201→   201→  return {
   202→   202→    firstMean,
   203→   203→    lastMean,
   204→   204→    first3,
   205→   205→    overallMean,
   206→   206→    fade,
   207→   207→    warmup,
   208→   208→    peak,
   209→   209→    peakSecond,
   210→   210→    low,
   211→   211→    errTotal,
   212→   212→    worstSec,
   213→   213→    worstSecErrs,
   214→   214→    worstWindowStart,
   215→   215→    worstWindowErrs,
   216→   216→  };
   217→   217→}

   240→   240→function hashStr(s: string): number {
   241→   241→  let h = 0;
   242→   242→  for (let i = 0; i < s.length; i++) h = (h * 31 + s.charCodeAt(i)) | 0;
   243→   243→  return Math.abs(h);
   244→   244→}

   250→   250→/** record: new PB with delta, first baseline, or a near-miss. */

   252→   252→  const { result, v } = c;
   253→   253→  if (result.wpm <= 0) return null;
   254→   254→  const prev = c.prevBest !== undefined ? c.prevBest : c.stats.personalBests[result.modeLabel];
   255→   255→  if (result.isPersonalBest) {
   256→   256→    if (prev && result.wpm > prev.wpm) {
   257→   257→      return {
   258→   258→        priority: 85,
   259→   259→        note: {
   260→   260→          kind: "record",
   261→   261→          title: v(["new personal best", "record falls"]),
   262→   262→          message: v([
   263→   263→            `${result.wpm} WPM for ${result.modeLabel}, up ${result.wpm - prev.wpm} from your previous best. Speed follows accuracy — you just proved it again.`,
   264→   264→            `+${result.wpm - prev.wpm} over your old ${result.modeLabel} best. Whatever pace you held in the clean middle stretch — that's the new default to train at.`,
   265→   265→          ]),
   266→   266→          metric: `+${result.wpm - prev.wpm} wpm`,
   267→   267→        },
   268→   268→      };
   269→   269→    }
   270→   270→    return {
   271→   271→      priority: 85,
   272→   272→      note: {
   273→   273→        kind: "record",
   274→   274→        title: "baseline set",
   275→   275→        message: `First score on record for ${result.modeLabel}: ${result.wpm} WPM at ${result.accuracy}%. Every test from here is measured against it.`,
   276→   276→        metric: `${result.wpm} wpm`,
   277→   277→      },
   278→   278→    };
   279→   279→  }
   280→   280→  if (prev && prev.wpm - result.wpm <= 2 && prev.wpm > result.wpm) {
   281→   281→    return {
   282→   282→      priority: 55,
   283→   283→      note: {
   284→   284→        kind: "record",
   285→   285→        title: "knocking on the record",
   286→   286→        message: `${prev.wpm - result.wpm === 0 ? "Tied pace with" : `Just ${prev.wpm - result.wpm} under`} your ${result.modeLabel} best (${prev.wpm}). One cleaner run today takes it.`,
   287→   287→        metric: `best ${prev.wpm}`,
   288→   288→      },
   289→   289→    };
   290→   290→  }
   291→   291→  return null;
   292→   292→}

   359→   359→/** trend: wpm vs last runs of the same label. */
   360→   360→function wpmTrendNote(c: DetectorCtx): Candidate | null {
   361→   361→  const { result, v } = c;
   362→   362→  const same = recentSameLabel(c.stats, result.modeLabel, result.id);
   363→   363→  if (same.length < 3) return null;
   364→   364→  const last5 = same.slice(0, 5);
   365→   365→  const avg = last5.reduce((a, b) => a + b.wpm, 0) / last5.length;
   366→   366→  const delta = Math.round(result.wpm - avg);
   367→   367→  if (delta >= 3) {
   368→   368→    return {
   369→   369→      priority: 60,
   370→   370→      note: {
   371→   371→        kind: "trend",
   372→   372→        title: v(["above your curve", "trending up"]),
   373→   373→        message: v([
   374→   374→          `+${delta} WPM over your last ${last5.length} ${result.modeLabel} runs (avg ${Math.round(avg)}). The training is compounding — same time tomorrow keeps the curve bent.`,
   375→   375→          `${delta} wpm better than your recent ${Math.round(avg)} average. Progress like this is made of boring, repeated, accurate reps — keep stacking them.`,
   376→   376→        ]),
   377→   377→        metric: `${delta >= 0 ? "+" : ""}${delta} wpm`,
   378→   378→      },
   379→   379→    };
   380→   380→  }
   381→   381→  if (delta <= -5) {
   382→   382→    return {
   383→   383→      priority: 62,
   384→   384→      note: {
   385→   385→        kind: "trend",
   386→   386→        title: "off-pace day",
   387→   387→        message: `${delta} WPM vs your recent ${Math.round(avg)} average. Off days are data: check whether accuracy dipped or one key cluster slowed the whole run.`,
   388→   388→        metric: `${delta} wpm`,
   389→   389→      },
   390→   390→    };
   391→   391→  }
   392→   392→  return null;
   393→   393→}

   436→   436→/** rhythm: cold open — slow first seconds before settling. */
   437→   437→function warmupNote(c: DetectorCtx): Candidate | null {
   438→   438→  const { result, curve, v } = c;
   439→   439→  if (!curve || result.duration < 20) return null;
   440→   440→  if (curve.warmup < 0.35) return null;
   441→   441→  return {
   442→   442→    priority: 54,
   443→   443→    note: {
   444→   444→      kind: "rhythm",
   445→   445→      title: "cold open",
   446→   446→      message: v([
   447→   447→        `The first seconds ran ~${Math.round(curve.first3)} wpm before settling at ~${Math.round(curve.overallMean)}. A short warmup test before your measured runs would lift every number on this screen.`,

   553→   553→function confusionNote(c: DetectorCtx): Candidate | null {
   554→   554→  const { ev, learning, v } = c;
   555→   555→  const combined = new Map<string, { expected: string; typed: string; total: number; thisTest: number }>();
   556→   556→  for (const c2 of learning.confusions) {
   557→   557→    const key = `${c2.expected}>${c2.typed}`;
   558→   558→    combined.set(key, { expected: c2.expected, typed: c2.typed, total: c2.count, thisTest: 0 });
   559→   559→  }
   560→   560→  for (const [key, n] of ev.confusions) {
   561→   561→    const [e, t] = key.split(">");
   562→   562→    const cur = combined.get(key);
   563→   563→    if (cur) {
   564→   564→      cur.total += n;
   565→   565→      cur.thisTest += n;
   566→   566→    } else {
   567→   567→      combined.set(key, { expected: e, typed: t, total: n, thisTest: n });
   568→   568→    }
   569→   569→  }
   570→   570→  let best: { expected: string; typed: string; total: number; thisTest: number } | null = null;
   571→   571→  for (const x of combined.values()) {
   572→   572→    if (x.total < 3 && x.thisTest < 2) continue;
   573→   573→    if (!best || x.total > best.total || (x.total === best.total && x.thisTest > best.thisTest)) best = x;
   574→   574→  }
   575→   575→  if (!best) return null;
   576→   576→  return {
   577→   577→    priority: 70,
   578→   578→    note: {
   579→   579→      kind: "tip",
   580→   580→      title: v([`finger habit: '${best.expected}' → '${best.typed}'`, `'${best.expected}' keeps turning into '${best.typed}'`]),
   581→   581→      message: v([
   582→   582→        `You've swapped '${best.expected}' for '${best.typed}' ${best.total}× recently${best.thisTest > 0 ? `, ${best.thisTest} in this very test` : ""}. That's a fixed finger-path habit — slow, deliberate reps on exactly that pair rewire it faster than volume does.`,
   583→   583→        `The scorecard says '${best.expected}' becomes '${best.typed}' under pressure (${best.total}×). Isolate it: type ten slow words containing '${best.expected}' before the next run.`,
   584→   584→      ]),
   585→   585→      metric: `×${best.total}${best.thisTest ? ` · ${best.thisTest} now` : ""}`,
   586→   586→    },
   587→   587→  };
   588→   588→}

   751→   751→/**
   752→   752→ * Generate human coaching insights after a completed test.
   753→   753→ * This is the "wants you to improve" personality of the app.
   754→   754→ *
   755→   755→ * `events` is the raw keystroke log of the finished test — the coach reads it
   756→   756→ * directly (which letters slipped, where pauses fell) instead of judging from
   757→   757→ * aggregates alone. `stats` should be the POST-record stats (so streak/history
   758→   758→ * notes reflect the test that just finished); `prevBest` is the personal best
   759→   759→ * BEFORE this test was recorded — pass it so the "new PB +delta" math works
   760→   760→ * (after recording, the PB already IS this result, and the delta would read 0).
   761→   761→ */
   762→   762→export function generateInsights(
   763→   763→  result: TestResult,
   764→   764→  events: CharEvent[],
   765→   765→  stats: StatsData,
   766→   766→  learning: LearningData,
   767→   767→  settings?: Settings,
   768→   768→  prevBest?: PersonalBest | null
   769→   769→): CoachInsight[] {
   770→   770→  const ev = analyzeEvents(events, result.focusKeys);
   771→   771→  const curve = analyzeCurve(result);
   772→   772→  const h = hashStr(result.id);
   773→   773→  const ctx: DetectorCtx = {
   774→   774→    result,
   775→   775→    ev,
   776→   776→    curve,
   777→   777→    stats,
   778→   778→    learning,
   779→   779→    settings,
   780→   780→    prevBest,
   781→   781→    v: <T,>(arr: T[]) => arr[h % arr.length],
   782→   782→  };
   783→   783→
   784→   784→  const candidates = [
   785→   785→    recordNote(ctx),
   786→   786→    errorTaxNote(ctx),
   787→   787→    accuracyNote(ctx),
   788→   788→    wpmTrendNote(ctx),
   789→   789→    accuracyTrendNote(ctx),
   790→   790→    fadeNote(ctx),
   791→   791→    warmupNote(ctx),
   792→   792→    errorSpikeNote(ctx),
   793→   793→    spreadNote(ctx),
   794→   794→    focusNote(ctx),
   795→   795→    charLossNote(ctx),
   796→   796→    confusionNote(ctx),
   797→   797→    transitionNote(ctx),
   798→   798→    slowKeysNote(ctx),
   799→   799→    hesitationNote(ctx),
   800→   800→    speedPushNote(ctx),
   801→   801→    recoveredNote(ctx),
   802→   802→    dueNote(ctx),
   803→   803→    streakNote(ctx),
   804→   804→    adaptiveNudgeNote(ctx),
   805→   805→  ].filter((x): x is Candidate => x !== null && x.note.message.length > 0);
   806→   806→
   807→   807→  candidates.sort((a, b) => b.priority - a.priority);
   808→   808→
   809→   809→  // kind caps: one note per kind (tips may double up), max 4 total, no dupes
   810→   810→  const picked: Candidate[] = [];
   811→   811→  const seen = new Set<string>();
   812→   812→  const kindCount = new Map<CoachInsight["kind"], number>();
   813→   813→  for (const cand of candidates) {
   814→   814→    const sig = cand.note.message;
   815→   815→    if (seen.has(sig)) continue;
   816→   816→    const n = kindCount.get(cand.note.kind) ?? 0;
   817→   817→    const cap = cand.note.kind === "tip" ? 2 : 1;
   818→   818→    if (n >= cap) continue;
   819→   819→    kindCount.set(cand.note.kind, n + 1);
   820→   820→    seen.add(sig);
   821→   821→    picked.push(cand);
   822→   822→    if (picked.length >= 4) break;
   823→   823→  }
   824→   824→  return picked.map((p) => p.note);
   825→   825→}

   831→   831→/** Idle-state line: what the engine will target next (adaptive mode only). */
   832→   832→export function nextTestPreview(learning: LearningData, mode: string): string {
   833→   833→  if (mode !== "adaptive") return "";
   834→   834→  if (learning.totalTests < 1) {
   835→   835→    return "Your first test calibrates the coach. From test 2 onward, every word is chosen for you.";
   836→   836→  }
   837→   837→  const weak = computeWeakKeys(learning).filter((w) => w.weakness > 0.08).slice(0, 2);
   838→   838→  const bgs = weakBigrams(learning, 3).filter((b) => b.weakness > 0.08).slice(0, 2);
   839→   839→
   840→   840→  const horizon = Date.now() + 48 * 3600 * 1000;
   841→   841→  let due = 0;
   842→   842→  for (const p of Object.values(learning.keyProfiles)) {
   843→   843→    if (p.mem && p.mem.reps > 0 && p.mem.due <= horizon) due += 1;
   844→   844→  }
   845→   845→  for (const p of Object.values(learning.bigramProfiles)) {
   846→   846→    if (p.mem && p.mem.reps > 0 && p.mem.due <= horizon) due += 1;
   847→   847→  }
   848→   848→
   849→   849→  const parts: string[] = [];
   850→   850→  if (weak.length) parts.push(`keys ${weak.map((w) => `'${w.key}'`).join(", ")}`);
   851→   851→  if (bgs.length) parts.push(`transitions ${bgs.map((b) => `'${b.bigram}'`).join(", ")}`);
   852→   852→
   853→   853→  if (!parts.length) {
   854→   854→    if (due >= 4) {
   855→   855→      return `${due} items are coming due for a refresher — the next test will start resurfacing them before they fade.`;
   856→   856→    }
   857→   857→    return "No strong weaknesses right now — the engine keeps mapping your fingers.";
   858→   858→  }
   859→   859→  const suffix = due >= 4 ? `, plus ${due} items due for a refresher` : "";
   860→   860→  return `Next test attacks your ${parts.join(" and ")}${suffix}.`;
   861→   861→}
